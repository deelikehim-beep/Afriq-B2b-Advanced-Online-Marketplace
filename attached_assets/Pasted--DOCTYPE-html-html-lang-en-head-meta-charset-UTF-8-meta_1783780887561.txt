// ======================== UTILITIES ========================
const fmt = (n) => '₦' + n.toLocaleString('en-NG');
const fmtShort = (n) => n >= 1000000 ? '₦' + (n/1000000).toFixed(1) + 'M' : n >= 1000 ? '₦' + (n/1000).toFixed(1) + 'K' : '₦' + n;

function showToast(msg, icon = 'fa-check-circle', type = 'success') {
    const t = document.getElementById('toast');
    const i = document.getElementById('toast-icon');
    const m = document.getElementById('toast-msg');
    m.textContent = msg;
    i.className = `fas ${icon} ${type === 'success' ? 'text-brand-400' : type === 'error' ? 'text-red-400' : 'text-gold-400'}`;
    t.classList.remove('hidden');
    t.classList.add('toast-enter');
    setTimeout(() => { t.classList.add('hidden'); t.classList.remove('toast-enter'); }, 2800);
}

function openModal(html) {
    document.getElementById('modal-content').innerHTML = html;
    document.getElementById('modal-overlay').classList.remove('hidden');
    document.body.style.overflow = 'hidden';
}

function closeModal(e) {
    if (!e || e.target.id === 'modal-overlay') {
        document.getElementById('modal-overlay').classList.add('hidden');
        document.body.style.overflow = '';
    }
}

function openPanel(html) {
    const panel = document.getElementById('fullscreen-panel');
    panel.innerHTML = html;
    panel.classList.remove('hidden');
    document.body.style.overflow = 'hidden';
}

function closePanel() {
    document.getElementById('fullscreen-panel').classList.add('hidden');
    document.body.style.overflow = '';
}

function updateBadges() {
    const cartCount = Store.cart.reduce((s, i) => s + i.qty, 0);
    const cartBadge = document.getElementById('cart-badge');
    if (cartCount > 0) { cartBadge.textContent = cartCount; cartBadge.classList.remove('hidden'); }
    else cartBadge.classList.add('hidden');

    const orderBadge = document.getElementById('order-badge');
    const activeOrders = Store.orders.filter(o => o.status !== 'completed').length;
    if (activeOrders > 0) { orderBadge.textContent = activeOrders; orderBadge.classList.remove('hidden'); }
    else orderBadge.classList.add('hidden');

    const msgBadge = document.getElementById('msg-badge');
    const unread = Store.conversations.reduce((s, c) => s + c.unread, 0);
    if (unread > 0) { msgBadge.textContent = unread; msgBadge.classList.remove('hidden'); }
    else msgBadge.classList.add('hidden');
}

// ======================== TABS ========================
function switchTab(tab) {
    Store.currentTab = tab;
    document.querySelectorAll('.nav-btn').forEach(btn => {
        const isActive = btn.dataset.tab === tab;
        btn.classList.toggle('tab-active', isActive);
        btn.classList.toggle('tab-inactive', !isActive);
        btn.querySelector('i').classList.toggle('text-brand-600', isActive);
    });
    if (tab === 'marketplace') renderMarketplace();
    else if (tab === 'orders') renderOrders();
    else if (tab === 'messages') renderMessages();
    else if (tab === 'wallet') renderWallet();
    else if (tab === 'profile') renderProfile();
    updateBadges();
}

function selectVariant(btn, name) {
    document.querySelectorAll('.variant-btn').forEach(b => {
        b.classList.remove('selected-variant', 'bg-brand-50', 'border-brand-500', 'text-brand-700');
        b.classList.add('bg-gray-100', 'border-transparent', 'text-gray-700');
    });
    btn.classList.remove('bg-gray-100', 'border-transparent', 'text-gray-700');
    btn.classList.add('selected-variant', 'bg-brand-50', 'border-brand-500', 'text-brand-700');
}

function adjustQty(delta, moq) {
    const input = document.getElementById('product-qty');
    let val = parseInt(input.value) + delta;
    if (val < moq) val = moq;
    input.value = val;
}

function addToCart(productId) {
    const p = Store.products.find(x => x.id === productId);
    const qty = parseInt(document.getElementById('product-qty')?.value || p.moq);
    const selectedVariant = document.querySelector('.selected-variant')?.textContent.trim() || p.variants[0].name;

    const existing = Store.cart.find(c => c.productId === productId && c.variant === selectedVariant);
    if (existing) {
        existing.qty = qty;
    } else {
        Store.cart.push({
            productId, name: p.name, price: p.price, qty, unit: p.unit,
            variant: selectedVariant, supplier: p.supplier, supplierId: p.supplierId,
            image: p.image, delivery: p.delivery, moq: p.moq
        });
    }

    updateBadges();
    closeModal();
    showToast('Added to cart', 'fa-shopping-cart');
}

function buyNow(productId) {
    addToCart(productId);
    setTimeout(() => showCart(), 300);
}

function updateCartQty(idx, delta) {
    const item = Store.cart[idx];
    item.qty += delta;
    if (item.qty < item.moq) item.qty = item.moq;
    showCart();
    updateBadges();
}

function removeFromCart(idx) {
    Store.cart.splice(idx, 1);
    showCart();
    updateBadges();
}

function confirmEscrowPayment() {
    const subtotal = Store.cart.reduce((s, i) => s + (i.price * i.qty), 0);
    const escrowFee = Math.round(subtotal * 0.025);
    const deliveryFee = Store.cart.length * 25000;
    const grandTotal = subtotal + escrowFee + deliveryFee;

    Store.user.balance -= grandTotal;
    Store.user.escrowLocked += subtotal;

    const orderId = 'ORD-' + new Date().toISOString().slice(2,10).replace(/-/g,'') + '-' + String(Store.orders.length + 1).padStart(3,'0');

    Store.cart.forEach(item => {
        const order = {
            id: orderId, productId: item.productId, productName: item.name,
            supplier: item.supplier, supplierId: item.supplierId,
            quantity: item.qty, unitPrice: item.price, totalAmount: item.price * item.qty,
            escrowFee: Math.round(item.price * item.qty * 0.025), grandTotal: item.price * item.qty + Math.round(item.price * item.qty * 0.025) + 25000,
            deliveryFee: 25000, variant: item.variant,
            status: "pending_payment", progress: 15, createdAt: "Just now",
            deliveryDate: "Pending", trackingId: "AFQ-TRK-" + Math.random().toString(36).substr(2,6).toUpperCase(),
            steps: [
                {name: "Order Placed", done: true, current: true},
                {name: "Payment Confirmed", done: false},
                {name: "Shipped", done: false},
                {name: "In Transit", done: false},
                {name: "Delivered", done: false},
                {name: "Completed", done: false}
            ]
        };
        Store.orders.unshift(order);

        // Add transaction
        Store.transactions.unshift({
            id: 'TXN-' + Date.now(), type: 'escrow_lock', amount: order.grandTotal,
            desc: `Escrow ${orderId} — ${item.name}`,
            date: 'Just now', status: 'pending', balance: Store.user.balance
        });

        // Notify supplier
        let conv = Store.conversations.find(c => c.supplierId === item.supplierId);
        if (conv) {
            conv.messages.push({
                sender: "me", type: "order",
                text: `New order ${orderId} for ${item.qty} ${item.unit} of ${item.name}. Payment secured in escrow.`,
                time: "Just now"
            });
            conv.lastMessage = `New order ${orderId} — ${item.name}`;
        }
    });

    Store.cart = [];
    updateBadges();
    closePanel();
    showToast('Escrow payment successful! Order placed.', 'fa-shield-alt');
    setTimeout(() => switchTab('orders'), 800);
}

function doConfirmDelivery(orderId) {
    const o = Store.orders.find(x => x.id === orderId);
    if (!o) return;

    o.status = 'completed';
    o.progress = 100;
    o.steps.forEach(s => { s.done = true; s.current = false; });

    Store.user.escrowLocked -= o.totalAmount;
    Store.transactions.unshift({
        id: 'TXN-' + Date.now(), type: 'release', amount: o.totalAmount,
        desc: `Released to ${o.supplier} — ${o.id}`,
        date: 'Just now', status: 'completed', balance: Store.user.balance
    });

    closeModal();
    closePanel();
    showToast('Delivery confirmed! Funds released to supplier.', 'fa-check-circle');
    updateBadges();
    setTimeout(() => switchTab('orders'), 500);
}

function sendMessage(convId) {
    const input = document.getElementById('chat-input');
    if (!input || !input.value.trim()) return;

    const c = Store.conversations.find(x => x.id === convId);
    c.messages.push({ sender: 'me', text: input.value, time: 'Just now', type: 'text' });
    c.lastMessage = input.value;
    c.time = 'Just now';

    openChat(convId);

    // Simulate reply
    setTimeout(() => {
        c.messages.push({ sender: 'them', text: 'Thanks for your message. We will get back to you shortly.', time: 'Just now', type: 'text' });
        c.lastMessage = 'Thanks for your message...';
        if (Store.currentTab !== 'messages') {
            c.unread = (c.unread || 0) + 1;
            updateBadges();
        }
    }, 2000);
}

function startChat(supplierId) {
    closeModal();
    closePanel();
    let c = Store.conversations.find(x => x.supplierId === supplierId);
    if (!c) {
        const p = Store.products.find(x => x.supplierId === supplierId);
        c = {
            id: Store.conversations.length + 1,
            with: p?.supplier || 'Supplier',
            avatar: (p?.supplier || 'SU').split(' ').map(x=>x[0]).join('').substring(0,2),
            online: true, supplierId,
            lastMessage: 'New conversation started', time: 'Just now', unread: 0,
            messages: []
        };
        Store.conversations.unshift(c);
    }
    setTimeout(() => openChat(c.id), 300);
}

function processDeposit() {
    const amount = parseInt(document.getElementById('deposit-amount')?.value || 0);
    if (amount < 1000) { showToast('Minimum deposit is ₦1,000', 'fa-exclamation-circle', 'error'); return; }

    Store.user.balance += amount;
    Store.transactions.unshift({
        id: 'TXN-' + Date.now(), type: 'deposit', amount,
        desc: 'Bank Transfer Deposit', date: 'Just now', status: 'completed', balance: Store.user.balance
    });
    closeModal();
    showToast(`₦${amount.toLocaleString()} deposited successfully!`, 'fa-wallet');
    switchTab('wallet');
}

// ======================== NOTIFICATIONS ========================
function showNotifications() {
    const unread = Store.notifications.filter(n => !n.read).length;

    openModal(`
        <div class="p-5">
            <div class="flex items-center justify-between mb-4">
                <h3 class="font-bold text-lg">Notifications</h3>
                ${unread > 0 ? `<button onclick="markAllRead()" class="text-xs text-brand-600 font-bold">Mark all read</button>` : ''}
            </div>
            <div class="space-y-2">
                ${Store.notifications.map(n => `
                    <div class="flex gap-3 p-3 rounded-xl ${n.read ? 'bg-white border border-gray-100' : 'bg-brand-50 border border-brand-100'} transition-all" onclick="markRead(${n.id})">
                        <div class="w-10 h-10 rounded-full ${n.color} flex items-center justify-center flex-shrink-0">
                            <i class="fas ${n.icon}"></i>
                        </div>
                        <div class="flex-1">
                            <div class="flex items-center justify-between mb-0.5">
                                <p class="text-xs font-bold ${n.read ? 'text-gray-700' : 'text-gray-900'}">${n.title}</p>
                                ${!n.read ? '<span class="w-2 h-2 bg-brand-500 rounded-full"></span>' : ''}
                            </div>
                            <p class="text-xs text-gray-500">${n.desc}</p>
                            <p class="text-[10px] text-gray-400 mt-1">${n.time}</p>
                        </div>
                    </div>
                `).join('')}
            </div>
            <button onclick="closeModal()" class="w-full py-3 text-sm text-gray-500 font-medium mt-4">Close</button>
        </div>
    `);
}

function markRead(id) {
    const n = Store.notifications.find(x => x.id === id);
    if (n) n.read = true;
    showNotifications();
    updateBadges();
}

function markAllRead() {
    Store.notifications.forEach(n => n.read = true);
    showNotifications();
    updateBadges();
}
